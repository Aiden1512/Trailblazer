cmake_minimum_required(VERSION 3.8)
project(trail_rerouting)
find_package(ament_cmake REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(rerouting_planner src/rerouting_planner.cpp)
target_compile_features(rerouting_planner PRIVATE cxx_std_17)
target_link_libraries(rerouting_planner yaml-cpp)
add_executable(bidirectional_inspection src/bidirectional_inspection.cpp)
target_compile_features(bidirectional_inspection PRIVATE cxx_std_17)
target_link_libraries(bidirectional_inspection yaml-cpp)
install(TARGETS rerouting_planner bidirectional_inspection DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
